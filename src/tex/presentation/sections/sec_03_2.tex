\begin{frame}{Backward Pass: Loss Function \& Phase 1}
    \textbf{The Loss Function}
    \begin{itemize}
        \justifying
        \item Training compares the rendered image with the original input image using a weighted combination of $\mathcal{L}_1$ loss and D-SSIM.
        \item $\mathcal{L}_1$ calculates the absolute pixel-wise error, while D-SSIM acts as a structural regularizer:
        $$\darkmathhighlight{ L = (1-\lambda)\cdot\mathcal{L}_1 + \lambda\cdot\mathcal{L}_{D-SSIM}, \quad \text{with } \lambda=0.2 }$$
    \end{itemize}

    \pause
    \vspace{1em}

    \textbf{Phase 1: Loss to Pixel Color Correction}
    \begin{itemize}
        \justifying
        \item The gradient calculates the required color correction per pixel ($C_p$).
        \item A positive gradient indicates an overestimation of color, negative indicates underestimation, and the magnitude dictates the degree of adjustment required:
        $$\darkmathhighlight{ \frac{\partial L}{\partial C_p} = \frac{1-\lambda}{N} \cdot sign(C_p - O_p) + \lambda \frac{\partial \mathcal{L}_{D-SSIM}}{\partial C_p} }$$
    \end{itemize}
\end{frame}

\begin{frame}{Backward Pass: Phase 2}
    \textbf{Phase 2: Pixel Color Correction to 2D Splat}
    \begin{itemize}
        \justifying
        \item The algorithm works alpha compositing backwards, flowing the pixel gradient into the evaluated opacity ($\alpha_i$) and color ($c_i$) of each 2D Gaussian in the tile.
    \end{itemize}

    \pause
    \vspace{0.5em}

    \begin{itemize}
        \justifying
        \item \textbf{Color Gradient:} Uses the chain rule to update the Gaussian's color based on its contribution (transmittance $T_i$ and opacity $\alpha_i$):
        $$ \text{with} \hspace{1mm} C_{p} = \sum_{i=1}^{N} c_i \alpha_i T_i: \hspace{4mm}
        \darkmathhighlight{
            \frac{\partial L}{\partial c_i} = 
            \frac{\partial L}{\partial C_p} \cdot \frac{\partial C_p}{\partial c_i} = 
            \frac{\partial L}{\partial C_p} \cdot (\alpha_i T_i)}  $$
    \end{itemize}

    \pause
    \vspace{0.5em}

    \begin{itemize}
        \justifying
        \item \textbf{Opacity Gradient:} Increases evaluated opacity if the Gaussian's color ($c_i$) reduces the pixel error more effectively than the accumulated background color behind it ($C_{behind}$):
        $$ \darkmathhighlight{
            \frac{\partial L}{\partial \alpha_i} = 
            \frac{\partial L}{\partial C_p} \cdot \frac{\partial C_p}{\partial \alpha_i} = 
            \frac{\partial L}{\partial C_p} \cdot T_i (c_i - C_{behind})} $$
    \end{itemize}
\end{frame}

\begin{frame}{Backward Pass: Phases 3 to 5}
    \textbf{Phase 3: 2D Splat to 2D Geometry}
    \begin{itemize}
        \justifying
        \item The opacity gradients are used to adjust the physical properties of the 2D Gaussians.
        \item This mathematically pulls the center position of the 2D Gaussian towards the pixel and stretches or contracts its 2D shape to minimize the error.
    \end{itemize}

    \pause
    \vspace{1em}

    \textbf{Phase 4: 2D Geometry to 3D Geometry}
    \begin{itemize}
        \justifying
        \item The GPU model changes from pixel-based to running per-Gaussian calculations.
        \item The 2D gradients are unprojected back into 3D space, translating the 2D adjustments into 3D center movements and 3D volume changes.
    \end{itemize}

    \pause
    \vspace{1em}

    \textbf{Phase 5: 3D Geometry to 3D Gaussian Attributes}
    \begin{itemize}
        \justifying
        \item The gradients flow backward through the 3D covariance matrix assembly.
        \item Adjustments are mapped directly into the raw, latent space variables for scale, rotation, and base opacity.
    \end{itemize}
\end{frame}

\begin{frame}{Adaptive Density Control \& Optimization}
    \textbf{Adaptive Density Control}
    \begin{itemize}
        \justifying
        \item Steps in every 100 iterations to counter over-reconstruction (covering too much geometry) and under-reconstruction (too few Gaussians).
        \item \textbf{Splitting:} Over-reconstructing Gaussians are split into two smaller ones.
        \item \textbf{Cloning:} Under-reconstructing Gaussians are cloned.
    \end{itemize}

    \pause
    \vspace{0.5em}

    \textbf{Pruning \& Opacity Reset}
    \begin{itemize}
        \justifying
        \item 3D Gaussians with extremely low opacity or unbound scale are routinely deleted.
        \item Every 3,000 iterations, the base opacity is reset to almost 0 for all Gaussians. This forces unimportant Gaussians to fade out completely and be pruned.
    \end{itemize}

    \pause
    \vspace{0.5em}

    \textbf{Optimization Step}
    \begin{itemize}
        \justifying
        \item Rather than standard gradient descent, 3DGS utilizes the Adam optimizer.
        \item Adam takes each parameter's history into account, smoothing out updates so that 3D Gaussians gently morph into their correct geometric shapes.
    \end{itemize}
\end{frame}
